# Απαραίτητες Βιβλιοθήκες

from pyspark.sql import functions
from pyspark.sql.types import StringType, FloatType
from pyspark.sql.functions import col, avg, month, year, date_format, max, min, format_number, year


#####################################################################################################


# Δημιουργία Dataframe για τα αρχεία agn.us.txt, ainv.us.txt, ale.us.txt

# Path για το αρχεία
agn_file_path = "dbfs:/FileStore/tables/agn_us.txt"
ainv_file_path = "dbfs:/FileStore/tables/ainv_us.txt"
ale_file_path = "dbfs:/FileStore/tables/ale_us.txt"

# Διάβασμα Αρχείων και Δημιουργία dataframe για τα τρία αρχεία
agn_df = spark.read.csv(agn_file_path, header=True)
ainv_df = spark.read.csv(ainv_file_path, header=True)
ale_df = spark.read.csv(ale_file_path, header=True)

# Κάνουμε τα παρακάτω στήλες τύπου float από string 
float_columns = ["Open", "High", "Low", "Close", "Volume", "OpenInt"]
for column in float_columns:
    agn_df = agn_df.withColumn(column, agn_df[column].cast(FloatType()))
    ainv_df = ainv_df.withColumn(column, ainv_df[column].cast(FloatType()))
    ale_df = ale_df.withColumn(column, ale_df[column].cast(FloatType()))

display(agn_df)
display(ainv_df)
display(ale_df)


#####################################################################################################


# Ποιος ήταν ο μέσος όρος για τις τιμές του ανοίγματος, κλεισίματος και όγκου συναλλαγών για κάθε ημερολογιακό μήνα για κάθε μετοχή;

# Μετοχή agn
agn_avg = agn_df.groupBy(month("Date").alias("Month")).agg(avg("Open").alias("Average Open"), avg("Close").alias("Average Close"), avg("Volume").alias("Average Volume")).orderBy("Month")

print("Ποιος ήταν ο μέσος όρος για τις τιμές του ανοίγματος, κλεισίματος και όγκου συναλλαγών για κάθε ημερολογιακό μήνα για την μεταχοή agn;")
agn_avg.show(truncate=False)

# Μετοχή ainv
ainv_avg = ainv_df.groupBy(month("Date").alias("Month")).agg(avg("Open").alias("Average Open"), avg("Close").alias("Average Close"), avg("Volume").alias("Average Volume")).orderBy("Month")

print("Ποιος ήταν ο μέσος όρος για τις τιμές του ανοίγματος, κλεισίματος και όγκου συναλλαγών για κάθε ημερολογιακό μήνα για την μεταχοή ainv;")
ainv_avg.show(truncate=False)

# Μετοχή ale
ale_avg = ale_df.groupBy(month("Date").alias("Month")).agg(avg("Open").alias("Average Open"), avg("Close").alias("Average Close"), avg("Volume").alias("Average Volume")).orderBy("Month")

print("Ποιος ήταν ο μέσος όρος για τις τιμές του ανοίγματος, κλεισίματος και όγκου συναλλαγών για κάθε ημερολογιακό μήνα για την μεταχοή ale;")
ale_avg.show(truncate=False)


#####################################################################################################


# Για πόσες ημέρες ήταν η τιμή του ανοίγματος της κάθε μετοχής πάνω από 35 δολάρια;

agn_open = agn_df.filter(col("Open") > 35.0000).count() 
print("Για πόσες ημέρες ήταν η τιμή του ανοίγματος της μετοχής agn πάνω από 35 δολάρια;") 
display(agn_open)
print()

ainv_open = ainv_df.filter(col("Open") > 35.0000).count() 
print("Για πόσες ημέρες ήταν η τιμή του ανοίγματος της μετοχής ainv πάνω από 35 δολάρια;")

ale_open = ale_df.filter(col("Open") > 35.0000).count()

# Ποιες ήταν οι μέρες με την υψηλότερη τιμή στο άνοιγμα και ποιες με την υψηλότερη τιμή στον όγκο συναλλαγών για κάθε μετοχή;

# agn: Εντοπίζουμε την μέγιστη τιμή του Open
agn_max_open = (
    agn_df
    .agg(max("Open").alias("Maximum Open"))
    .join(agn_df, col("Open") == col("Maximum Open"))
    .select("Date", "Maximum Open")
)
print("Η μέρα με την υψηλότερη τιμή στο άνοιγμα για την μετοχή agn:")
agn_max_open.show(truncate=False)

# agn: Εντοπίζουμε την μέγιστη τιμή του Volume
agn_max_volume = (
    agn_df
    .agg(max("Volume").alias("Maximum Volume"))
    .join(agn_df, col("Volume") == col("Maximum Volume"))
    .select("Date", format_number("Maximum Volume", 0).alias("Maximum Volume"))
)
print("Η μέρα με την υψηλότερη τιμή στον όγκο συναλλαγών για την μετοχή agn:")
agn_max_volume.show(truncate=False)

# ainv: Εντοπίζουμε την μέγιστη τιμή του Volume

# Ποιες ήταν οι χρονιές που κάθε μετοχή σημείωσε την υψηλότερη τιμή στο άνοιγμα και ποιες και τη χαμηλότερη τιμή στο κλείσιμο;
